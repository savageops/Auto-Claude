Selection State (from useIssueSelection.ts):

```typescript
const [enabledConflicts, setEnabledConflicts] = useState<Set<string>>(new Set());

const toggleConflict = useCallback((filePath: string) => {
  setEnabledConflicts(prev => {
    const next = new Set(prev);
    if (next.has(filePath)) {
      next.delete(filePath);
    } else {
      next.add(filePath);
    }
    return next;
  });
}, []);
```
